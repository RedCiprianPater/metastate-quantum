# --- IBM transitive pins (added 2026-08 · unblocks Render sdist build) ------
# Without these, pip's resolver backtracks through ibm-platform-services
# looking for a version whose declared dep tree is fully satisfiable. It
# ends up on 0.55.2 (sdist-only for py-generic), whose setup.py does
# `import pkg_resources` and fails inside pip's isolated build env because
# recent setuptools (>= 70) doesn't ship pkg_resources by default in build
# isolation. Pinning to the wheel versions pip was already willing to accept
# (visible in the failed build log) skips the broken sdist entirely.
ibm-cloud-sdk-core==3.20.6
ibm-platform-services==0.56.0

# --- Local Aer simulator (NEW · v1.2.0-osaka) -------------------------------
# Required for two reasons:
#   1. The `simulator` backend and universal fallback when a hardware backend
#      is unconfigured or returns an error.
#   2. The Osaka hybrid path: when n_qubits > 1, qubit 0 runs on the real
#      ¹⁷¹Yb⁺ ion, and the tail is computed on Aer, returning
#      hardware_status = "live-hybrid (Osaka 1q QPU + sim tail)".
#
# 0.15.1 is the bugfix release of the Aer 0.15 line that pairs with Qiskit
# 1.2.x. Aer 0.16 requires Qiskit 1.3, and Aer 0.17 requires Qiskit 2.x —
# neither is compatible with the pinned qiskit==1.2.4 above.
qiskit-aer==0.15.1
