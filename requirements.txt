# --- Web framework (existing pins, kept verbatim) ---------------------------
fastapi==0.115.0

# --- Cross-cutting HTTP client pin (added 2026-08 · resolves urllib3 collision) ---
# Four packages in this tree touch urllib3 with mutually-exclusive constraints
# if we let each resolve independently:
#
#   qiskit-ibm-runtime 0.30.0  →  urllib3>=1.21.1               (lower bound only)
#   ibm-cloud-sdk-core 3.20.x  →  urllib3>=2.1.0,<3.0.0         (v2 only)
#   oqtopus-client 1.1.x       →  urllib3>=1.25.3,<2.1.0        (v1 or v2.0.x)
#   quri-parts-oqtopus 1.1.3   →  urllib3                        (any)
#
# The intersection of the middle two is empty, so pip errors out. Pinning
# urllib3 here to <2.1.0 satisfies oqtopus-client + qiskit-ibm-runtime, and
# forces pip to backtrack ibm-cloud-sdk-core (and its parent ibm-platform-services)
# to older releases that still accepted urllib3<2.1.0 (3.19.x line and earlier).
# We deliberately do NOT pin ibm-cloud-sdk-core or ibm-platform-services here —
# the urllib3 pin cascades the correct resolution automatically, and pip's
# --prefer-binary flag in render.yaml keeps it on wheels.
urllib3>=1.26.0,<2.1.0

# --- IBM Quantum backend (existing pins, kept verbatim) ---------------------
qiskit==1.2.4
qiskit-ibm-runtime==0.30.0

# --- Local Aer simulator (NEW · v1.2.0-osaka) -------------------------------
# Required for two reasons:
#   1. The `simulator` backend and universal fallback when a hardware backend
#      is unconfigured or returns an error.
#   2. The Osaka hybrid path: when n_qubits > 1, qubit 0 runs on the real
#      ¹⁷¹Yb⁺ ion, and the tail is computed on Aer, returning
#      hardware_status = "live-hybrid (Osaka 1q QPU + sim tail)".
#
# 0.15.1 is the bugfix release of the Aer 0.15 line that pairs with Qiskit
# 1.2.x. Aer 0.16 requires Qiskit 1.3, and Aer 0.17 requires Qiskit 2.x —
# neither is compatible with the pinned qiskit==1.2.4 above.
qiskit-aer==0.15.1
