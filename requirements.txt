# --- HTTP client pin — resolves the urllib3 collision ------------------------
# Four packages in this tree touch urllib3. The strictest common window is:
#   qiskit-ibm-runtime 0.30.0  →  urllib3>=1.21.1
#   ibm-cloud-sdk-core 3.16.7  →  urllib3<2.0.0,>=1.26.0    ← we target this side
#   oqtopus-client 1.1.x       →  urllib3>=1.25.3,<2.1.0
#   quri-parts-oqtopus 1.1.3   →  urllib3                    (any)
# Intersection: [1.26.0, 2.0.0). We pin to that window.
urllib3>=1.26.0,<2.0.0

# --- IBM chain — explicit pre-urllib3-v2 pins (both are wheels on PyPI) ------
# The newer ibm-cloud-sdk-core line (3.20.x) demands urllib3>=2.1.0, which
# collides with oqtopus-client. The newer ibm-platform-services (0.55.x+)
# transitively demands that sdk-core, and worse, 0.55.2 ships as sdist-only,
# whose setup.py imports pkg_resources and blows up in pip's isolated build
# env on modern setuptools.
#
# 3.16.7 is the last sdk-core version to declare urllib3<2.0.0 explicitly,
# and it ships ibm_cloud_sdk_core-3.16.7-py3-none-any.whl (verified on PyPI).
# 0.44.0 is the ibm-platform-services from Sept 2023 that pairs with sdk-core
# 3.16.x. Being ~6 months older than the urllib3 v2 transition guarantees it
# won't demand a newer sdk-core, and it ships a real wheel. This combo avoids
# the sdist path entirely — pip never even considers 0.55.2.
ibm-cloud-sdk-core==3.16.7
ibm-platform-services==0.44.0

uvicorn[standard]==0.30.6
